import { test, expect } from '@playwright/test';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Step 1: The user clicks the 'Measurement' button in the toolbar to open the measurement panel.
    const measurementToggle = page.getByTestId('measurement-toggle');
    await measurementToggle.click();

    // Verify the measurement panel is visible
    await expect(page.getByTestId('measurement-panel')).toBeVisible();

    // Step 2 & 3: The user clicks several points on the map canvas to draw a line, then double-clicks to finish.
    // We need to interact with the map container. We'll pick coordinates that are likely within the visible map area.
    // Since we don't know the exact viewport, we'll try to click relative to the map container's bounding box.
    const mapContainer = page.getByTestId('map-container');
    const box = await mapContainer.boundingBox();

    if (!box) {
        throw new Error('Map container bounding box not found');
    }

    // Define points to draw a simple line.
    // Point 1: Start near the center-left
    const point1 = { x: box.x + box.width * 0.3, y: box.y + box.height * 0.5 };
    // Point 2: Middle
    const point2 = { x: box.x + box.width * 0.5, y: box.y + box.height * 0.5 };
    // Point 3: End near the center-right
    const point3 = { x: box.x + box.width * 0.7, y: box.y + box.height * 0.5 };

    // Click first point
    await page.mouse.click(point1.x, point1.y);
    
    // Click second point
    await page.mouse.click(point2.x, point2.y);
    
    // Double-click to finish the measurement
    await page.mouse.dblclick(point3.x, point3.y);

    // Wait for the measurement result to appear and settle
    // The measurement panel should display a length value with a unit.
    // We'll poll for the presence of text that looks like a measurement (e.g., "123 m", "1.2 km")
    // inside the measurement panel.
    await expect.poll(async () => {
        const panel = page.getByTestId('measurement-panel');
        const text = await panel.textContent();
        // Check if there's any text that resembles a number followed by a unit
        return text ? /[\d.]+\s*(m|km|mi|ft)/.test(text) : false;
    }).toBe(true);
});
