import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Step 1: Click the visibility toggle of the UV-Index overlay layer.
    // The UI map indicates the layer switcher is visible by default and contains the operational layers.
    // We use force: true because Chakra UI checkbox controls have a visually hidden input.
    await page.getByTestId('layer-switcher').getByRole('checkbox', { name: 'UV-Index' }).click({ force: true });

    // Verify the toggle is checked
    await expect(page.getByTestId('layer-switcher').getByRole('checkbox', { name: 'UV-Index' })).toBeChecked();

    // Step 2: Wait for the map to load the layer tiles and verify it is rendered.
    // We poll the helper function which reads from the map model to confirm the layer is visible.
    await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
