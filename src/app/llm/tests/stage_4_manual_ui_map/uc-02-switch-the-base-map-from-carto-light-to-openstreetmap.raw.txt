import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle } from '../../map-model-helpers';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready and the default base layer to be loaded
  await expect.poll(() => getActiveBaseLayerTitle(page)).resolves.toBe('Carto Light');

  // 1. The user opens the base map selector in the layer switcher.
  // The layer switcher is visible by default. We look for the dropdown control.
  const layerSwitcher = page.getByTestId('layer-switcher');
  // The base map selector is a dropdown within the layer switcher.
  // We assume the dropdown is accessible by role or label.
  // If not explicitly labeled, we might need to find the select element.
  // Based on the UI map, it's a dropdown with options.
  const baseMapDropdown = layerSwitcher.getByRole('combobox', { name: /base map/i });
  await baseMapDropdown.click();

  // 2. The user selects 'OpenStreetMap' as the base map.
  await page.getByRole('option', { name: 'OpenStreetMap' }).click();

  // Expected result: The OpenStreetMap base map is selected.
  await expect.poll(() => getActiveBaseLayerTitle(page)).resolves.toBe('OpenStreetMap');

  // Expected result: The Carto Light base map is no longer selected.
  await expect.poll(() => getActiveBaseLayerTitle(page)).resolves.not.toBe('Carto Light');
});
