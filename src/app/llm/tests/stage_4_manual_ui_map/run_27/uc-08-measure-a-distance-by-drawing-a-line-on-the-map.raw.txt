import { test, expect } from '@playwright/test';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Step 1: The user clicks the 'Measurement' button in the toolbar to open the measurement panel.
  const measurementToggle = page.getByTestId('measurement-toggle');
  await measurementToggle.click();

  // Verify the measurement panel is visible
  await expect(page.getByTestId('measurement-panel')).toBeVisible();

  // Step 2: The user clicks several points on the map canvas to draw a line.
  // We need to click on the map container. The map is interactive.
  // We will click three points to form a simple line.
  const mapContainer = page.getByTestId('map-container');
  
  // Click first point (approximate center-left)
  await mapContainer.click({ position: { x: 100, y: 100 } });
  
  // Click second point (approximate center-right)
  await mapContainer.click({ position: { x: 200, y: 100 } });
  
  // Click third point (approximate center-bottom)
  await mapContainer.click({ position: { x: 150, y: 200 } });

  // Step 3: The user double-clicks to finish the measurement.
  // Double-click on the last point or nearby to finish.
  await mapContainer.dblclick({ position: { x: 150, y: 200 } });

  // Expected results:
  // - The measurement panel is visible. (Already checked above, but good to ensure it stays visible)
  await expect(page.getByTestId('measurement-panel')).toBeVisible();

  // - The measurement panel displays a length value with a unit.
  // We look for the measurement element inside the panel.
  const measurementElement = page.getByTestId('measurement');
  await expect(measurementElement).toBeVisible();

  // The measurement element likely contains text with a number and a unit (e.g., "1.5 km").
  // We assert that it contains some text that looks like a measurement.
  // Since we can't predict the exact value, we check for the presence of a number and a unit.
  // Common units might be "km", "m", "mi", "ft".
  // We use a regex to match a number followed by a unit.
  await expect(measurementElement).toHaveText(/[\d.,]+\s*(km|m|mi|ft|cm|mm)/i);
});
