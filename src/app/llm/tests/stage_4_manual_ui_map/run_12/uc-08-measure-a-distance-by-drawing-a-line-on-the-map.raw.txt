import { test, expect } from '@playwright/test';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // 1. Click the measurement toggle to open the measurement panel
  const measurementToggle = page.getByTestId('measurement-toggle');
  await measurementToggle.click();

  // Wait for the measurement panel to become visible
  await expect(page.getByTestId('measurement-panel')).toBeVisible();

  // 2. Click several points on the map canvas to draw a line
  // We need to click on the map container. The map is interactive via the canvas.
  // We'll click at distinct coordinates to create a line.
  const mapContainer = page.getByTestId('map-container');

  // First point
  await mapContainer.click({ position: { x: 200, y: 200 } });
  // Second point
  await mapContainer.click({ position: { x: 300, y: 300 } });
  // Third point
  await mapContainer.click({ position: { x: 400, y: 200 } });

  // 3. Double-click to finish the measurement
  await mapContainer.dblclick({ position: { x: 400, y: 200 } });

  // Expected results:
  // - The measurement panel is visible (already asserted above, but good to reaffirm)
  await expect(page.getByTestId('measurement-panel')).toBeVisible();

  // - The measurement panel displays a length value with a unit.
  // The measurement element inside the panel should contain text that looks like a number followed by a unit (e.g., "1.23 km", "500 m")
  const measurementElement = page.getByTestId('measurement');
  await expect(measurementElement).toBeVisible();

  // Check for a length value with a unit.
  // We use a regex to match a number (possibly decimal) followed by a unit.
  // Common units in mapping apps are km, m, mi, ft.
  await expect(measurementElement).toContainText(/(\d+\.?\d*)\s*(km|m|mi|ft)/);
});
