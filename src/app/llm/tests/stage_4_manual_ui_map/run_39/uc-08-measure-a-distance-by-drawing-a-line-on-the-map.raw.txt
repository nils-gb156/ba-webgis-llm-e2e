import { test, expect } from '@playwright/test';
import { getMapZoomLevel } from '../../../map-model-helpers';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    const mapContainer = page.getByTestId('map-container');
    const measurementToggle = page.getByTestId('measurement-toggle');
    const measurementPanel = page.getByTestId('measurement-panel');
    const measurement = page.getByTestId('measurement');

    await expect(mapContainer).toBeVisible();
    await expect(measurementToggle).toBeVisible();

    await expect.poll(async () => typeof (await getMapZoomLevel(page))).toBe('number');

    if (!(await measurementPanel.isVisible())) {
        await measurementToggle.click();
    }

    await expect(measurementPanel).toBeVisible();
    await expect(measurement).toBeVisible();

    const box = await mapContainer.boundingBox();
    if (!box) {
        throw new Error('Map container has no bounding box.');
    }

    const points = [
        { x: Math.round(box.width * 0.62), y: Math.round(box.height * 0.42) },
        { x: Math.round(box.width * 0.72), y: Math.round(box.height * 0.50) },
        { x: Math.round(box.width * 0.66), y: Math.round(box.height * 0.60) },
        { x: Math.round(box.width * 0.56), y: Math.round(box.height * 0.66) }
    ];

    await mapContainer.click({ position: points[0] });
    await mapContainer.click({ position: points[1] });
    await mapContainer.click({ position: points[2] });
    await mapContainer.dblclick({ position: points[3] });

    await expect.poll(async () => {
        const text = (await measurement.textContent())?.replace(/\s+/g, ' ').trim() ?? '';
        const match = text.match(
            /(\d+(?:[.,]\d+)?)\s?(mm|cm|m|km|meter|meters|kilometer|kilometers)\b/i
        );
        return match ? Number(match[1].replace(',', '.')) : -1;
    }).toBeGreaterThan(0);

    await expect.poll(async () => {
        const text = (await measurement.textContent())?.replace(/\s+/g, ' ').trim() ?? '';
        const match = text.match(
            /(\d+(?:[.,]\d+)?)\s?(mm|cm|m|km|meter|meters|kilometer|kilometers)\b/i
        );
        return match?.[2] ?? '';
    }).toMatch(/^(mm|cm|m|km|meter|meters|kilometer|kilometers)$/i);
});
