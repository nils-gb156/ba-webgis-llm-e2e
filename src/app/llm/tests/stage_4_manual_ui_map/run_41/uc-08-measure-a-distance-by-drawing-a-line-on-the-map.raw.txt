import { test, expect } from '@playwright/test';
import { getMapCenter } from '../../../map-model-helpers';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    const mapContainer = page.getByTestId('map-container');
    const measurementToggle = page.getByTestId('measurement-toggle');
    const measurementPanel = page.getByTestId('measurement-panel');
    const measurementContent = page.getByTestId('measurement');

    await expect(mapContainer).toBeVisible();
    await expect(measurementToggle).toBeVisible();
    await expect.poll(() => getMapCenter(page)).toBeTruthy();

    if (!(await measurementPanel.isVisible())) {
        await measurementToggle.click();
    }

    await expect(measurementPanel).toBeVisible();
    await expect(measurementContent).toBeVisible();

    const box = await mapContainer.boundingBox();
    expect(box).not.toBeNull();

    if (!box) {
        throw new Error('Map container has no bounding box.');
    }

    await mapContainer.click({
        position: { x: box.width * 0.68, y: box.height * 0.35 }
    });
    await mapContainer.click({
        position: { x: box.width * 0.78, y: box.height * 0.45 }
    });
    await mapContainer.click({
        position: { x: box.width * 0.72, y: box.height * 0.58 }
    });
    await mapContainer.dblclick({
        position: { x: box.width * 0.84, y: box.height * 0.68 }
    });

    await expect(measurementPanel).toBeVisible();
    await expect(measurementContent).toContainText(/\d+(?:[.,]\d+)?\s?(mm|cm|m|km)\b/i);
});
