import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle } from '../../../map-model-helpers';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    const layerSwitcher = page.getByTestId('layer-switcher');

    await expect(layerSwitcher).toBeVisible();
    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('Carto Light');

    const comboboxSelector = layerSwitcher.getByRole('combobox').first();

    if ((await comboboxSelector.count()) > 0) {
        const tagName = await comboboxSelector.evaluate((element) => element.tagName.toLowerCase());

        if (tagName === 'select') {
            await comboboxSelector.selectOption({ label: 'OpenStreetMap' });
        } else {
            await comboboxSelector.click();

            const option = page.getByRole('option', { name: 'OpenStreetMap', exact: true }).first();
            const menuItemRadio = page.getByRole('menuitemradio', { name: 'OpenStreetMap', exact: true }).first();
            const radio = page.getByRole('radio', { name: 'OpenStreetMap', exact: true }).first();
            const button = page.getByRole('button', { name: 'OpenStreetMap', exact: true }).first();

            if ((await option.count()) > 0) {
                await option.click();
            } else if ((await menuItemRadio.count()) > 0) {
                await menuItemRadio.click();
            } else if ((await radio.count()) > 0) {
                await radio.click({ force: true });
                await expect(radio).toBeChecked();
            } else {
                await button.click();
            }
        }
    } else {
        const baseMapButton = layerSwitcher.getByRole('button', { name: /Carto Light/ }).first();
        await expect(baseMapButton).toBeVisible();
        await baseMapButton.click();

        const option = page.getByRole('option', { name: 'OpenStreetMap', exact: true }).first();
        const menuItemRadio = page.getByRole('menuitemradio', { name: 'OpenStreetMap', exact: true }).first();
        const radio = page.getByRole('radio', { name: 'OpenStreetMap', exact: true }).first();
        const button = page.getByRole('button', { name: 'OpenStreetMap', exact: true }).first();

        if ((await option.count()) > 0) {
            await option.click();
        } else if ((await menuItemRadio.count()) > 0) {
            await menuItemRadio.click();
        } else if ((await radio.count()) > 0) {
            await radio.click({ force: true });
            await expect(radio).toBeChecked();
        } else {
            await button.click();
        }
    }

    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('OpenStreetMap');
    await expect
        .poll(async () => (await getActiveBaseLayerTitle(page)) === 'Carto Light')
        .toBe(false);
});
