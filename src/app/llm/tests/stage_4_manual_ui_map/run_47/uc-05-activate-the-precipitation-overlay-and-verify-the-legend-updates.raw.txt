import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 5: Activate the Precipitation overlay and verify the legend updates', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Step 1: Click the visibility toggle of the Precipitation overlay layer.
  // The layer switcher is visible by default. We locate the checkbox for "Precipitation".
  // Using force: true because Chakra UI renders the real input visually hidden.
  const precipitationToggle = page.getByRole('checkbox', { name: 'Precipitation' });
  await precipitationToggle.click({ force: true });

  // Step 2: Verify the Precipitation overlay is actually rendered on the map.
  await expect.poll(() => isLayerRendered(page, 'Precipitation')).toBe(true);

  // Step 2: Verify the legend displays an entry corresponding to the Precipitation layer.
  // The legend is visible by default. We look for text inside the legend container.
  const legend = page.getByTestId('legend');
  await expect(legend).toContainText('Precipitation');
});
