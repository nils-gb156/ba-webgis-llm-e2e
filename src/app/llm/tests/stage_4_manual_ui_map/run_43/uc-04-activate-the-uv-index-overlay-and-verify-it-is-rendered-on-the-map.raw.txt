import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Step 1: Click the visibility toggle of the UV-Index overlay layer.
  // The UI map indicates the layer switcher is visible by default.
  // The UV-Index layer is in the operational list. We locate the checkbox for "UV-Index".
  // Since Chakra UI checkboxes have a hidden input, we use force: true.
  const uvIndexCheckbox = page.getByRole('checkbox', { name: 'UV-Index' });
  await uvIndexCheckbox.click({ force: true });

  // Verify the toggle is in the enabled (checked) state
  await expect(uvIndexCheckbox).toBeChecked();

  // Step 2: Wait for the map to load the layer tiles and verify it is rendered.
  // We use the helper function isLayerRendered wrapped in expect.poll to wait for the map state to settle.
  await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
