import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 7: Click both point station layers to show feature info', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Verify preconditions: Layers are active
  await expect.poll(() => isLayerRendered(page, 'UV-Index Stations')).toBe(true);
  await expect.poll(() => isLayerRendered(page, 'EUCOS Ground Stations')).toBe(true);

  // Step 1: Click on the map at the specified coordinates
  const targetX = 1188692.84;
  const targetY = 6767643.28;
  
  // Click the map container to trigger GetFeatureInfo
  await page.getByTestId('map-container').click({
    position: { x: targetX, y: targetY }
  });

  // Step 2: Wait for the info panel to load station info for both layers
  // The info panel is visible by default, but the specific sections appear asynchronously
  
  // Wait for UV-Index Station info section to appear
  await expect(page.getByTestId('uvi-station-section')).toBeVisible();
  
  // Wait for EUCOS Ground Station info section to appear
  await expect(page.getByTestId('eucos-station-section')).toBeVisible();

  // Verify that the info panel displays feature information for both layers
  // The sections should contain some content indicating feature info is loaded
  await expect(page.getByTestId('uvi-station-info')).toBeVisible();
  await expect(page.getByTestId('eucos-station-info')).toBeVisible();
});
