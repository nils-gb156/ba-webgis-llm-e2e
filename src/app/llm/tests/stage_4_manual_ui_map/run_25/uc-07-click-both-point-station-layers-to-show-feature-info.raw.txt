import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 7: Click both point station layers to show feature info', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Ensure the target operational layers are active
    await expect.poll(() => isLayerRendered(page, 'UV-Index Stations')).toBe(true);
    await expect.poll(() => isLayerRendered(page, 'EUCOS Ground Stations')).toBe(true);

    // Click on the map at the specified coordinates where both station types overlap
    await page.click('[data-testid="map-container"]', {
        position: { x: 1188692.84, y: 6767643.28 }
    });

    // Wait for and assert the UV-Index Station info section is visible
    await expect(page.getByTestId('uvi-station-section')).toBeVisible();
    await expect(page.getByTestId('uvi-station-info')).toBeVisible();

    // Wait for and assert the EUCOS Ground Station info section is visible
    await expect(page.getByTestId('eucos-station-section')).toBeVisible();
    await expect(page.getByTestId('eucos-station-info')).toBeVisible();
});
