import { test, expect } from '@playwright/test';

test('Use Case 5: Activate the Precipitation overlay and verify the legend updates', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const precipitationToggle = page.getByRole('checkbox', { name: /^Precipitation$/i });
  const legend = page
    .getByRole('region', { name: /^Legend$/i })
    .or(page.getByRole('complementary', { name: /^Legend$/i }))
    .or(page.getByLabel(/^Legend$/i))
    .first();

  await expect(page.getByText('Precipitation', { exact: true })).toBeVisible();
  await expect(legend).toBeVisible();
  await expect(precipitationToggle).not.toBeChecked();

  await precipitationToggle.click({ force: true });

  await expect(precipitationToggle).toBeChecked();
  await expect(legend.getByText('Precipitation', { exact: true })).toBeVisible();
});
