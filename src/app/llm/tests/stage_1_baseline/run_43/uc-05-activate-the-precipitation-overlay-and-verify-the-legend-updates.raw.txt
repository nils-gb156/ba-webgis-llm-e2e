import { test, expect } from '@playwright/test';

test('Use Case 5: Activate the Precipitation overlay and verify the legend updates', async ({
  page
}) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');
  await page.waitForLoadState('domcontentloaded');

  const precipitationCheckbox = page.getByRole('checkbox', {
    name: 'Precipitation',
    exact: true
  });
  const precipitationSwitch = page.getByRole('switch', {
    name: 'Precipitation',
    exact: true
  });
  const precipitationToggle =
    (await precipitationCheckbox.count()) > 0 ? precipitationCheckbox : precipitationSwitch;

  await expect(precipitationToggle).toBeVisible();

  const legendHeading = page.getByRole('heading', { name: 'Legend', exact: true });
  const legendRegion = page.getByRole('region', { name: 'Legend', exact: true });
  const legendGroup = page.getByRole('group', { name: 'Legend', exact: true });
  const legendComplementary = page.getByRole('complementary', {
    name: 'Legend',
    exact: true
  });

  let legendContainer = legendRegion;
  if ((await legendContainer.count()) === 0) {
    legendContainer = legendGroup;
  }
  if ((await legendContainer.count()) === 0) {
    legendContainer = legendComplementary;
  }

  if ((await legendHeading.count()) > 0) {
    await expect(legendHeading).toBeVisible();
  } else {
    await expect(legendContainer).toBeVisible();
  }

  await expect(precipitationToggle).not.toBeChecked();

  await precipitationToggle.click({ force: true });

  await expect(precipitationToggle).toBeChecked();

  if ((await legendContainer.count()) > 0) {
    await expect(legendContainer.getByText('Precipitation', { exact: true })).toBeVisible();
  } else {
    await expect
      .poll(async () => {
        return await page.getByText('Precipitation', { exact: true }).count();
      })
      .toBeGreaterThanOrEqual(2);
  }
});
