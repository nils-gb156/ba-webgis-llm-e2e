import { test, expect } from '@playwright/test';

test('Use Case 5: Activate the Precipitation overlay and verify the legend updates', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const precipitationToggle = page.getByRole('checkbox', { name: 'Precipitation', exact: true });
  const legendHeading = page.getByRole('heading', { name: 'Legend', exact: true });

  await expect(precipitationToggle).toBeVisible();
  await expect(legendHeading).toBeVisible();
  await expect(precipitationToggle).not.toBeChecked();

  const visiblePrecipitationTextCount = async () => {
    return await page.getByText('Precipitation', { exact: true }).evaluateAll((elements) =>
      elements.filter((element) => {
        const htmlElement = element as HTMLElement;
        const style = window.getComputedStyle(htmlElement);
        const rect = htmlElement.getBoundingClientRect();
        return style.display !== 'none' && style.visibility !== 'hidden' && rect.width > 0 && rect.height > 0;
      }).length
    );
  };

  const initialVisiblePrecipitationTextCount = await visiblePrecipitationTextCount();

  await precipitationToggle.click({ force: true });
  await expect(precipitationToggle).toBeChecked();

  let legendContainer = page.getByRole('region').filter({ has: legendHeading });
  if ((await legendContainer.count()) === 0) {
    legendContainer = page.getByRole('group').filter({ has: legendHeading });
  }
  if ((await legendContainer.count()) === 0) {
    legendContainer = page.getByRole('complementary').filter({ has: legendHeading });
  }

  if ((await legendContainer.count()) > 0) {
    const scopedLegend = legendContainer.first();
    await expect(scopedLegend).toBeVisible();
    await expect(scopedLegend.getByText('Precipitation', { exact: true })).toBeVisible();
  } else {
    await expect.poll(visiblePrecipitationTextCount).toBeGreaterThan(initialVisiblePrecipitationTextCount);
  }
});
