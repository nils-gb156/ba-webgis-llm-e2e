import { test, expect } from '@playwright/test';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the application to load and the layer switcher to be visible
  await expect(page.getByTestId('layer-switcher')).toBeVisible();

  // Step 1: Open the base map selector in the layer switcher
  // Assuming the base map selector is a button or toggle within the layer switcher
  const baseMapSelector = page.getByTestId('layer-switcher-base-map-selector');
  await baseMapSelector.click();

  // Wait for the base map options to appear
  await expect(page.getByRole('option', { name: 'OpenStreetMap' })).toBeVisible();

  // Step 2: Select 'OpenStreetMap' as the base map
  await page.getByRole('option', { name: 'OpenStreetMap' }).click();

  // Expected results:
  // - The OpenStreetMap base map is selected.
  // - The Carto Light base map is no longer selected.
  // We assert that the OpenStreetMap option is now marked as selected
  await expect(page.getByRole('option', { name: 'OpenStreetMap' })).toBeChecked();

  // And that Carto Light is no longer selected
  await expect(page.getByRole('option', { name: 'Carto Light' })).not.toBeChecked();
});
