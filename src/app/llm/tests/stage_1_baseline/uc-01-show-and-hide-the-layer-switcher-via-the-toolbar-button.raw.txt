import { test, expect } from '@playwright/test';

test('Use Case 1: Show and hide the layer switcher via the toolbar button', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the application to load and the layer switcher to be initially visible
  const layerSwitcherButton = page.getByRole('button', { name: 'Layer Switcher' });
  await expect(layerSwitcherButton).toBeVisible();

  // Identify the layer switcher panel. Assuming it has a test id or accessible name.
  // Since test ids are not automatic, we look for a panel that might be associated with the button.
  // Often, such panels have a specific role or label. Let's assume the panel is a dialog or a region.
  // If no specific test id is known, we might rely on the button's aria-expanded or similar,
  // but Playwright's expect is auto-retrying. Let's try to find the panel by its likely content or role.
  // A common pattern is a panel with role="region" or similar, or a specific test id like 'layer-switcher'.
  // Given the constraints, let's assume there is a test id 'layer-switcher' for the panel,
  // or we can look for a container that appears/disappears.
  // If no test id is available, we might need to infer from the button's action.
  // However, the prompt says "The layer switcher (TOC) is initially visible".
  // Let's assume the panel has a test id 'layer-switcher' for robustness, as is common in such apps.
  // If not, we might use a more generic locator. Let's try getByTestId first.
  const layerSwitcherPanel = page.getByTestId('layer-switcher');

  // Step 1: Click the 'Layer Switcher' button to hide the panel.
  await layerSwitcherButton.click();

  // Expected result: After the first click, the layer switcher panel is no longer visible.
  await expect(layerSwitcherPanel).not.toBeVisible();

  // Step 2: Click the 'Layer Switcher' button again to show the panel.
  await layerSwitcherButton.click();

  // Expected result: After the second click, the layer switcher panel is visible again.
  await expect(layerSwitcherPanel).toBeVisible();
});
