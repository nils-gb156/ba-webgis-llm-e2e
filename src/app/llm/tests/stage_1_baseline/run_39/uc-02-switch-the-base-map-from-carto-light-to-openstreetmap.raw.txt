import { test, expect } from '@playwright/test';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const cartoLightOption = page.getByRole('radio', { name: 'Carto Light', exact: true });
  const openStreetMapOption = page.getByRole('radio', { name: 'OpenStreetMap', exact: true });
  const baseMapSelectorToggle = page.getByRole('button', { name: /base ?maps?/i });

  await expect(baseMapSelectorToggle).toBeVisible();

  if (!(await openStreetMapOption.isVisible())) {
    await baseMapSelectorToggle.click();
  }

  await expect(cartoLightOption).toBeVisible();
  await expect(openStreetMapOption).toBeVisible();

  await expect(cartoLightOption).toBeChecked();
  await expect(openStreetMapOption).not.toBeChecked();

  await openStreetMapOption.click({ force: true });

  await expect(openStreetMapOption).toBeChecked();
  await expect(cartoLightOption).not.toBeChecked();
});
