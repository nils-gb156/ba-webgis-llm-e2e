import { test, expect } from '@playwright/test';
import { getHighlightedCoordinate, isLayerRendered } from '../../../map-model-helpers';

test('Use Case 6: Click a map position to show the weather forecast', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Ensure map is ready
    await expect(page.getByTestId('map-container')).toBeVisible();

    // Click on the map canvas to trigger GetFeatureInfo / weather forecast lookup
    // Using the center of the map container as a safe default click position
    const mapBox = await page.getByTestId('map-container').boundingBox();
    if (!mapBox) {
        throw new Error('Map container not found');
    }
    await page.getByTestId('map-container').click({
        position: { x: mapBox.width / 2, y: mapBox.height / 2 }
    });

    // Wait for the info panel to be visible (it is visible by default, but we wait for content to load)
    await expect(page.getByTestId('info-panel')).toBeVisible();

    // Wait for the weather forecast section to appear in the info panel
    await expect(page.getByTestId('weather-forecast-section')).toBeVisible();

    // Assert that the clicked position is highlighted on the map
    await expect.poll(() => getHighlightedCoordinate(page)).toBeTruthy();

    // Assert that the forecast contains 24 entries
    // The weather-forecast-entry elements are children of the weather-forecast-section or similar container
    // We count the number of forecast entries visible
    const forecastEntries = page.getByTestId(/weather-forecast-entry-\d+/);
    await expect(forecastEntries).toHaveCount(24);
});
