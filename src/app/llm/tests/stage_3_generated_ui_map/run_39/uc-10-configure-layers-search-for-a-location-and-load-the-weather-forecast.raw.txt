import { test, expect } from "@playwright/test";
import { isLayerRendered, getMapCenter } from "../../../map-model-helpers";

test("Use Case 10: Configure layers, search for a location and load the weather forecast", async ({
  page,
}) => {
  await page.goto(
    "http://localhost:5173/ba-webgis-llm-e2e/",
  );

  // Wait for the map and initial layers to be ready
  await expect.poll(() => isLayerRendered(page, "Temperature")).toBe(true);

  // 1. Hide Temperature overlay
  // The layer switcher is visible by default. We locate the Temperature layer
  // within the layer switcher panel and click its checkbox to toggle it off.
  const layerSwitcher = page.getByRole("panel", { name: /layer/i });
  const temperatureToggle = layerSwitcher.getByRole("checkbox", {
    name: "Temperature",
  });
  await temperatureToggle.click({ force: true });

  // 2. Show Precipitation overlay
  // Precipitation is initially hidden. We click its checkbox to enable it.
  const precipitationToggle = layerSwitcher.getByRole("checkbox", {
    name: "Precipitation",
  });
  await precipitationToggle.click({ force: true });

  // Assert layer states: Temperature should be hidden, Precipitation should be visible
  await expect.poll(() => isLayerRendered(page, "Temperature")).toBe(false);
  await expect
    .poll(() => isLayerRendered(page, "Precipitation"))
    .toBe(true);

  // 3. Search for a location
  const geocoderInput = page.getByTestId("geocoder-input");
  await geocoderInput.fill("Münster");

  // 4. Wait for results and select the first one
  // The results panel appears below the input. We wait for the first result item.
  const firstResult = page.getByTestId("geocoder-result-item-0");
  await expect(firstResult).toBeVisible();
  await firstResult.click();

  // 5. Wait for the map to navigate to the selected location
  // We poll the map center to ensure it has moved from the initial extent.
  const initialCenter = await getMapCenter(page);
  await expect.poll(() => getMapCenter(page)).not.toEqual(initialCenter);

  // 6. Wait for the info panel to load the weather forecast
  // The info panel is visible by default. We look for the weather forecast section
  // and assert that it contains the expected number of entries (24).
  const weatherForecastSection = page.getByTestId("weather-forecast-section");
  await expect(weatherForecastSection).toBeVisible();

  // Poll for the forecast entries to appear, as the forecast data loads asynchronously
  await expect
    .poll(async () => {
      const entries = page.getByTestId("weather-forecast-entry");
      return entries.count();
    })
    .toBe(24);
});
