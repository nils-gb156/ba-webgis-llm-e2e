import { test, expect } from '@playwright/test';
import { getHighlightedCoordinate, isLayerRendered } from '../../../map-model-helpers';

test('Use Case 6: Click a map position to show the weather forecast', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Ensure info panel is visible (it is visible by default, but let's be explicit)
  await expect(page.getByTestId('info-panel')).toBeVisible();

  // Click on the map center to trigger the weather forecast
  await page.getByTestId('map-container').click({ position: { x: 100, y: 100 } });

  // Wait for the info panel to load the forecast
  // The forecast section should appear
  await expect(page.getByTestId('weather-forecast-section')).toBeVisible({ timeout: 10000 });

  // Check that the clicked position is highlighted on the map
  // We poll the highlighted coordinate to ensure it has been set
  await expect.poll(() => getHighlightedCoordinate(page)).toBeTruthy();

  // Check that the forecast contains 24 entries
  // We count the number of weather-forecast-entry elements
  const forecastEntries = page.getByTestId(/weather-forecast-entry-\d+/);
  await expect(forecastEntries).toHaveCount(24);
});
