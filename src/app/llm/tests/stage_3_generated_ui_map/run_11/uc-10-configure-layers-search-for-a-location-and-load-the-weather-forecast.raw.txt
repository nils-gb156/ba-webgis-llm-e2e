import { test, expect } from '@playwright/test';
import { getMapCenter } from '../../../map-model-helpers';

test('Use Case 10: Configure layers, search for a location and load the weather forecast', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Wait for the map and initial layers to be ready
    await expect(page.getByTestId('map-container')).toBeVisible();
    await expect(page.getByTestId('layer-switcher')).toBeVisible();
    await expect(page.getByTestId('info-panel')).toBeVisible();

    // 1. Hide the Temperature overlay layer.
    // The layer switcher is visible by default. We find the Temperature layer entry.
    // Assuming the layer switcher uses a list structure where each item has a test id or label.
    // Since specific layer toggle test IDs are not in the UI map, we use getByRole with exact name.
    // We need to scope this to the layer-switcher panel to avoid ambiguity.
    const layerSwitcher = page.getByTestId('layer-switcher');
    const temperatureToggle = layerSwitcher.getByRole('checkbox', { name: 'Temperature', exact: true });
    await expect(temperatureToggle).toBeChecked(); // Precondition: initially visible/checked
    await temperatureToggle.click({ force: true }); // Toggle off

    // 2. Show the Precipitation overlay layer.
    const precipitationToggle = layerSwitcher.getByRole('checkbox', { name: 'Precipitation', exact: true });
    await expect(precipitationToggle).not.toBeChecked(); // Precondition: initially hidden/unchecked
    await precipitationToggle.click({ force: true }); // Toggle on

    // 3. Search for 'Münster' using the geocoder.
    const geocoderInput = page.getByTestId('geocoder-input');
    await geocoderInput.click();
    await geocoderInput.fill('Münster');

    // 4. Wait for results and select the first one.
    // The results appear in an element with test id geocoder-results.
    // We wait for the first result item to be visible.
    const firstResult = page.getByTestId('geocoder-result-item-0');
    await expect(firstResult).toBeVisible();
    await firstResult.click();

    // 5. Wait for the map to navigate to the selected location.
    // We poll the map center to ensure it has changed from the initial position.
    // We don't know the exact center, but we can assert that the map is no longer at the starting center
    // or simply that a center exists and is stable. A better heuristic for "navigated" is usually
    // waiting for the geocoder results to disappear or a specific map load event, but here we rely on
    // the map model. We'll poll until the center is defined and potentially different from a known start,
    // or just wait for the info panel to update which implies navigation.
    // Let's wait for the info panel to start showing loading or new content.
    // Since we need to assert map navigation, we check if the center is no longer undefined.
    await expect.poll(() => getMapCenter(page)).toBeTruthy();

    // 6. Wait for the info panel to load the forecast.
    // The expected result is 24 weather forecast entries.
    // We look for the weather-forecast-section and count the entries.
    const weatherSection = page.getByTestId('weather-forecast-section');
    await expect(weatherSection).toBeVisible();
    
    // Poll for the number of forecast entries to be 24.
    await expect.poll(async () => {
        const entries = page.getByTestId('weather-forecast-entry');
        return await entries.count();
    }).toBe(24);

    // Additional assertions from Expected Results:
    // - Precipitation toggle is disabled (checked/enabled state).
    await expect(precipitationToggle).toBeChecked();
    // - Temperature toggle is enabled (checked state).
    await expect(temperatureToggle).toBeChecked();
});
