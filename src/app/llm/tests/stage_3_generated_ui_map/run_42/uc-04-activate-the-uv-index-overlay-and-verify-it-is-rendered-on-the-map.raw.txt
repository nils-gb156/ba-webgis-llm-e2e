import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle, isLayerRendered } from '../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({
    page
}) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    const mapContainer = page.getByTestId('map-container');
    const layerSwitcher = page.getByTestId('layer-switcher');
    const uvIndexLabel = layerSwitcher.getByText('UV-Index', { exact: true });
    const uvIndexCheckbox = layerSwitcher.getByRole('checkbox', {
        name: 'UV-Index',
        exact: true
    });

    await expect(mapContainer).toBeVisible();
    await expect(layerSwitcher).toBeVisible();
    await expect(uvIndexLabel).toBeVisible();
    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('Carto Light');
    await expect(uvIndexCheckbox).not.toBeChecked();
    await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(false);

    const uvIndexLayerRequests: string[] = [];
    const matchesUvIndexLayerRequest = (url: string): boolean => {
        let decodedUrl = url;
        try {
            decodedUrl = decodeURIComponent(url);
        } catch {
            decodedUrl = url;
        }

        return (
            !/stations/i.test(decodedUrl) &&
            (/(uv[-_ ]?index)/i.test(decodedUrl) || /\buvi\b/i.test(decodedUrl)) &&
            /(GetMap|wms|wmts|tiles|\.png|\.jpe?g|\.webp)/i.test(decodedUrl)
        );
    };

    page.on('request', (request) => {
        if (
            ['image', 'xhr', 'fetch'].includes(request.resourceType()) &&
            matchesUvIndexLayerRequest(request.url())
        ) {
            uvIndexLayerRequests.push(request.url());
        }
    });

    await uvIndexCheckbox.click({ force: true });

    await expect(uvIndexCheckbox).toBeChecked();
    await expect.poll(() => uvIndexLayerRequests.length > 0).toBe(true);
    await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
