import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready before interacting
  await expect(page.getByTestId('map-container')).toBeVisible();

  // 1. The user clicks the visibility toggle of the UV-Index overlay layer to show it.
  // The layer switcher is visible by default. We locate the UV-Index layer item and click its checkbox.
  // Using force: true because Chakra UI checkboxes have a hidden input under a decorative element.
  const uvIndexLayerItem = page.getByTestId('layer-switcher').getByText('UV-Index').first();
  const uvIndexCheckbox = uvIndexLayerItem.getByRole('checkbox', { name: 'UV-Index' });
  
  await uvIndexCheckbox.click({ force: true });

  // 2. The user waits for the map to load the layer tiles.
  // We verify the UV-Index overlay layer toggle is in the enabled (checked) state.
  await expect(uvIndexCheckbox).toBeChecked();

  // And verify that the UV-Index overlay tiles are rendered on the map canvas.
  await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
