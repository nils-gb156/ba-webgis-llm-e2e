import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 5: Activate the Precipitation overlay and verify the legend updates', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready and initial layers to settle
  await expect.poll(() => isLayerRendered(page, 'Temperature')).toBe(true);

  // Step 1: Click the visibility toggle of the Precipitation overlay layer.
  // The layer switcher is visible by default. We locate the Precipitation layer
  // within the layer switcher panel and click its checkbox.
  const layerSwitcher = page.getByRole('panel', { name: 'Layer Switcher' });
  const precipitationCheckbox = layerSwitcher.getByRole('checkbox', { name: 'Precipitation' });
  
  // Precipitation is initially hidden, so we click to enable it.
  // Using force: true because Chakra UI checkboxes render the input visually hidden.
  await precipitationCheckbox.click({ force: true });

  // Verify the toggle is now in the enabled (checked) state.
  await expect(precipitationCheckbox).toBeChecked();

  // Verify the Precipitation layer is actually rendered on the map.
  await expect.poll(() => isLayerRendered(page, 'Precipitation')).toBe(true);

  // Step 2: View the legend and verify it displays an entry for the Precipitation layer.
  // The legend is visible by default. We check for the presence of the precipitation legend element.
  const precipitationLegend = page.getByTestId('precipitation-legend');
  await expect(precipitationLegend).toBeVisible();
});
