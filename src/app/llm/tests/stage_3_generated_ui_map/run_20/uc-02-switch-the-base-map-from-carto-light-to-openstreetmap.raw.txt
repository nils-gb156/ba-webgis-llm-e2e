import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle } from '../../../map-model-helpers';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    const layerSwitcher = page.getByTestId('layer-switcher');
    await expect(layerSwitcher).toBeVisible();

    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('Carto Light');

    const osmRadio = layerSwitcher.getByRole('radio', { name: 'OpenStreetMap', exact: true });
    const cartoLightRadio = layerSwitcher.getByRole('radio', { name: 'Carto Light', exact: true });
    const baseMapSelectorToggle = layerSwitcher.getByRole('button', { name: /base maps?|basemaps?/i });

    if (!(await osmRadio.isVisible()) && (await baseMapSelectorToggle.count()) > 0) {
        await baseMapSelectorToggle.first().click();
    }

    if ((await osmRadio.count()) > 0) {
        await expect(osmRadio).toBeVisible();
        await osmRadio.click({ force: true });
        await expect(osmRadio).toBeChecked();
        if ((await cartoLightRadio.count()) > 0) {
            await expect(cartoLightRadio).not.toBeChecked();
        }
    } else {
        const osmButton = layerSwitcher.getByRole('button', { name: 'OpenStreetMap', exact: true });
        const osmLabeledControl = layerSwitcher.getByLabel('OpenStreetMap', { exact: true });

        if ((await osmButton.count()) > 0) {
            await osmButton.click();
        } else if ((await osmLabeledControl.count()) > 0) {
            await osmLabeledControl.click({ force: true });
        } else {
            await layerSwitcher.getByText('OpenStreetMap', { exact: true }).click();
        }
    }

    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('OpenStreetMap');
    await expect.poll(() => getActiveBaseLayerTitle(page)).not.toBe('Carto Light');
});
