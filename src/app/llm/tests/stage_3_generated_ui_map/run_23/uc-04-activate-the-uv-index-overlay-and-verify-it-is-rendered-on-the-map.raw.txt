import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Wait for the map to be ready before interacting
    await expect(page.getByTestId('map-container')).toBeVisible();

    // The UV-Index layer is initially hidden.
    // We need to find its toggle in the layer switcher.
    // Based on the UI map, the layer switcher is visible by default.
    // We look for the UV-Index entry in the layer switcher and click its checkbox.
    // Since specific test ids for layer items are not provided in the UI map,
    // we use getByRole with the accessible name "UV-Index" scoped to the layer switcher.
    const layerSwitcher = page.getByTestId('layer-switcher');
    const uvIndexCheckbox = layerSwitcher.getByRole('checkbox', { name: 'UV-Index' });

    // Click the checkbox to enable the layer.
    // Chakra UI checkboxes require force: true as per conventions.
    await uvIndexCheckbox.click({ force: true });

    // Verify the toggle is now checked
    await expect(uvIndexCheckbox).toBeChecked();

    // Verify the UV-Index layer is rendered on the map canvas
    await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
