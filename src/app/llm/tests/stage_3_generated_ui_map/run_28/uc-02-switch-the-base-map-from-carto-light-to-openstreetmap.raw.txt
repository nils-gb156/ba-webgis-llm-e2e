import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle } from '../../../map-model-helpers';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    const layerSwitcher = page.getByTestId('layer-switcher');
    await expect(layerSwitcher).toBeVisible();

    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('Carto Light');

    const openStreetMapOption = layerSwitcher.getByRole('radio', {
        name: 'OpenStreetMap',
        exact: true
    });
    const cartoLightOption = layerSwitcher.getByRole('radio', {
        name: 'Carto Light',
        exact: true
    });

    if (!(await openStreetMapOption.isVisible().catch(() => false))) {
        const baseMapSelectorToggle = layerSwitcher.getByRole('button', {
            name: /base maps?|basemaps?|background maps?|background layers?|base layers?/i
        });

        if (await baseMapSelectorToggle.count()) {
            const toggle = baseMapSelectorToggle.first();
            const expanded = await toggle.getAttribute('aria-expanded');
            if (expanded === 'false') {
                await toggle.click();
            }
        }
    }

    await expect(openStreetMapOption).toBeVisible();
    await expect(cartoLightOption).toBeVisible();
    await expect(cartoLightOption).toBeChecked();
    await expect(openStreetMapOption).not.toBeChecked();

    await openStreetMapOption.click({ force: true });

    await expect(openStreetMapOption).toBeChecked();
    await expect(cartoLightOption).not.toBeChecked();
    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('OpenStreetMap');
    await expect.poll(() => getActiveBaseLayerTitle(page)).not.toBe('Carto Light');
});
