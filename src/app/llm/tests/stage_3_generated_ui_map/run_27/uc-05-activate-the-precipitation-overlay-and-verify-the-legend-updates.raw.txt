import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 5: Activate the Precipitation overlay and verify the legend updates', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Wait for the map and layers to be ready
    await expect(page.getByTestId('layer-switcher')).toBeVisible();
    await expect(page.getByTestId('legend')).toBeVisible();

    // Step 1: Click the visibility toggle of the Precipitation overlay layer
    // The layer switcher is visible by default. We need to find the Precipitation layer entry.
    // Based on the UI map, we don't have a specific test id for the layer item,
    // so we use getByRole with the layer name.
    const precipitationLayerToggle = page.getByRole('checkbox', { name: 'Precipitation' });
    await expect(precipitationLayerToggle).toBeChecked({ checked: false });
    await precipitationLayerToggle.click({ force: true });

    // Verify the Precipitation overlay layer toggle is in the enabled (checked) state
    await expect(precipitationLayerToggle).toBeChecked();

    // Verify the layer is actually rendered on the map
    await expect.poll(() => isLayerRendered(page, 'Precipitation')).toBe(true);

    // Step 2: View the legend
    // The legend should update to reflect the newly active layer.
    // We look for the precipitation legend element.
    const precipitationLegend = page.getByTestId('precipitation-legend');
    await expect(precipitationLegend).toBeVisible();
});
