import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready and initial layers to load
  await expect(page.getByTestId('map-container')).toBeVisible();
  await expect.poll(() => isLayerRendered(page, 'Temperature')).toBe(true);

  // Locate the UV-Index layer in the layer switcher and click its visibility toggle.
  // The layer switcher is visible by default. We scope the search to the layer switcher panel
  // to avoid ambiguity with other UI elements.
  const layerSwitcher = page.getByTestId('layer-switcher');
  const uvIndexToggle = layerSwitcher.getByRole('checkbox', { name: 'UV-Index' });

  // Click the checkbox toggle to enable the layer.
  // Chakra UI checkboxes render the real input visually hidden; use force: true.
  await uvIndexToggle.click({ force: true });

  // Assert that the toggle is now in the checked (enabled) state.
  await expect(uvIndexToggle).toBeChecked();

  // Assert that the UV-Index layer is now rendered on the map canvas.
  await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
