import { test, expect } from '@playwright/test';
import { getHighlightedCoordinate, isLayerRendered } from '../../../map-model-helpers';

test('Use Case 6: Click a map position to show the weather forecast', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Ensure the info panel is visible by default
  await expect(page.getByTestId('info-panel')).toBeVisible();

  // Click a position on the map canvas
  const mapContainer = page.getByTestId('map-container');
  await mapContainer.click({ position: { x: 300, y: 300 } });

  // Wait for the info panel to load the forecast
  // The forecast section should appear
  await expect(page.getByTestId('weather-forecast-section')).toBeVisible();

  // Verify the clicked position is highlighted on the map
  // We use expect.poll to wait for the highlight to settle
  await expect.poll(() => getHighlightedCoordinate(page)).toBeTruthy();

  // Verify the forecast contains 24 entries
  const forecastEntries = page.getByTestId(/weather-forecast-entry/);
  await expect.poll(async () => {
    const count = await forecastEntries.count();
    return count;
  }).toBe(24);
});
