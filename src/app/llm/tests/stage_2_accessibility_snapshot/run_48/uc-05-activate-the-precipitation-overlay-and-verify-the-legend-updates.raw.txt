import { test, expect } from '@playwright/test';

test('Use Case 5: Activate the Precipitation overlay and verify the legend updates', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Step 1: Click the visibility toggle of the Precipitation overlay layer
  // The layer switcher is already open (Layer Switcher button is pressed)
  await page.getByRole('checkbox', { name: 'Precipitation' }).click({ force: true });

  // Step 2: Verify the Precipitation overlay layer toggle is in the enabled (checked) state
  await expect(page.getByRole('checkbox', { name: 'Precipitation' })).toBeChecked();

  // Step 2: Verify the legend displays an entry corresponding to the Precipitation layer
  // The legend panel is already open (Legend Switcher button is pressed)
  // We check for the presence of a precipitation-related legend item.
  // Since we don't have the exact text, we look for a new legend item that wasn't there before,
  // or specifically look for "Precipitation" in the legend.
  await expect(page.getByRole('heading', { name: 'Precipitation', exact: true })).toBeVisible();
});
