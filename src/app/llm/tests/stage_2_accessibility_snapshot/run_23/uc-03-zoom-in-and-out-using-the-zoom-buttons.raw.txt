import { test, expect } from '@playwright/test';

test('Use Case 3: Zoom in and out using the zoom buttons', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const zoomInButton = page.getByTestId('zoom-in-button');
  const zoomOutButton = page.getByTestId('zoom-out-button');
  const scaleViewer = page.getByTestId('scale-viewer');

  await expect(zoomInButton).toBeVisible();
  await expect(zoomOutButton).toBeVisible();
  await expect(scaleViewer).toBeVisible();

  const readScaleDenominator = async () => {
    const text = await scaleViewer.innerText();
    const match = text.match(/1\s+to\s+([\d.,]+)/i);
    return match ? Number(match[1].replace(/[.,]/g, '')) : undefined;
  };

  const initialScale = await expect
    .poll(async () => await readScaleDenominator())
    .not.toBeUndefined();

  const initialScaleValue = (await readScaleDenominator()) as number;

  await zoomInButton.click();

  await expect
    .poll(async () => await readScaleDenominator())
    .toBeLessThan(initialScaleValue);

  const zoomedInScaleValue = (await readScaleDenominator()) as number;

  await zoomOutButton.click();

  await expect
    .poll(async () => await readScaleDenominator())
    .toBeGreaterThan(zoomedInScaleValue);
});
