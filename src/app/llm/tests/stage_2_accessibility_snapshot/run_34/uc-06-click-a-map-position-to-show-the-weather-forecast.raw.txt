import { test, expect } from '@playwright/test';

test('Use Case 6: Click a map position to show the weather forecast', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const infoPanel = page.getByTestId('info-panel');
  const infoPanelToggle = page.getByTestId('info-panel-toggle');
  const weatherForecastSection = page.getByTestId('weather-forecast-section');
  const mapContainer = page.getByTestId('map-container');

  await expect(infoPanel).toBeVisible();
  await expect(infoPanelToggle).toHaveAttribute('aria-pressed', 'true');
  await expect(weatherForecastSection).toBeVisible();
  await expect(
    weatherForecastSection.getByText('Click on the map to load a forecast.', { exact: true })
  ).toBeVisible();
  await expect(mapContainer).toBeVisible();

  const mapBox = await mapContainer.boundingBox();
  if (!mapBox) {
    throw new Error('Map container is not ready for interaction.');
  }

  await mapContainer.click({
    position: {
      x: Math.round(mapBox.width * 0.5),
      y: Math.round(mapBox.height * 0.5)
    }
  });

  await expect(
    weatherForecastSection.getByText('Click on the map to load a forecast.', { exact: true })
  ).not.toBeVisible();

  await expect(weatherForecastSection).toBeVisible();

  await expect.poll(async () => {
    const listItems = await weatherForecastSection.getByRole('listitem').count();
    const articles = await weatherForecastSection.locator('article').count();
    const buttons = await weatherForecastSection.getByRole('button').count();
    const tableRows = await weatherForecastSection.locator('tbody tr').count();
    const images = await weatherForecastSection.getByRole('img').count();

    return Math.max(listItems, articles, buttons, tableRows, images);
  }).toBe(24);
});
