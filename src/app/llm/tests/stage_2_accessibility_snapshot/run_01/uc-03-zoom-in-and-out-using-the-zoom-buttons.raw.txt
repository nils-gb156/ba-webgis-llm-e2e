import { test, expect } from '@playwright/test';

test('Use Case 3: Zoom in and out using the zoom buttons', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const zoomInButton = page.getByTestId('zoom-in-button');
  const zoomOutButton = page.getByTestId('zoom-out-button');
  const scaleViewer = page.getByTestId('scale-viewer');

  const getScaleDenominator = async (): Promise<number | undefined> => {
    const text = await scaleViewer.textContent();
    if (!text) {
      return undefined;
    }

    const matches = text.match(/([\d.,]+)(?!.*[\d.,])/);
    if (!matches) {
      return undefined;
    }

    const numericText = matches[1].replace(/[^\d]/g, '');
    if (!numericText) {
      return undefined;
    }

    const value = Number.parseInt(numericText, 10);
    return Number.isNaN(value) ? undefined : value;
  };

  await expect(page.getByRole('application', { name: 'webgis map' })).toBeVisible();
  await expect(zoomInButton).toBeVisible();
  await expect(zoomOutButton).toBeVisible();
  await expect(scaleViewer).toBeVisible();

  await expect.poll(getScaleDenominator).toBeGreaterThan(0);
  const initialScale = await getScaleDenominator();
  expect(initialScale).toBeDefined();

  await zoomInButton.click();

  await expect.poll(getScaleDenominator).toBeLessThan(initialScale!);
  const zoomedInScale = await getScaleDenominator();
  expect(zoomedInScale).toBeDefined();

  await zoomOutButton.click();

  await expect.poll(getScaleDenominator).toBeGreaterThan(zoomedInScale!);
});
