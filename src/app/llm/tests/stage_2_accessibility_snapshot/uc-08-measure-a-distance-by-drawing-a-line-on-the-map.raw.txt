import { test, expect } from '@playwright/test';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready and interactive
  const mapContainer = page.getByTestId('map-container');
  await expect(mapContainer).toBeVisible();
  await expect(mapContainer).toBeInViewport();

  // Step 1: Click the 'Measurement' button in the toolbar
  const measurementToggle = page.getByRole('button', { name: 'Measurement' });
  await measurementToggle.click();

  // Step 2: Click several points on the map canvas to draw a line
  // We need to get the bounding box of the map to click on it
  const mapBox = await mapContainer.boundingBox();
  if (!mapBox) {
    throw new Error('Map container not found or not visible');
  }

  // Define points to draw a line (relative to the map container)
  // Point 1: Center-ish
  const point1X = mapBox.x + mapBox.width / 2;
  const point1Y = mapBox.y + mapBox.height / 2;

  // Point 2: Top-right
  const point2X = mapBox.x + mapBox.width * 0.75;
  const point2Y = mapBox.y + mapBox.height * 0.25;

  // Point 3: Bottom-left
  const point3X = mapBox.x + mapBox.width * 0.25;
  const point3Y = mapBox.y + mapBox.height * 0.75;

  // Click point 1
  await page.mouse.click(point1X, point1Y);
  // Small delay to ensure click is registered
  await page.waitForTimeout(100);

  // Click point 2
  await page.mouse.click(point2X, point2Y);
  await page.waitForTimeout(100);

  // Click point 3
  await page.mouse.click(point3X, point3Y);
  await page.waitForTimeout(100);

  // Step 3: Double-click to finish the measurement
  await page.mouse.dblclick(point3X, point3Y);

  // Expected result: The measurement panel is visible and displays a length value with a unit
  // The measurement result is likely shown in the info panel or a dedicated measurement result area
  // Based on the context, the info panel is open. Let's look for text that looks like a measurement.
  // We'll poll for a pattern that resembles a distance measurement (e.g., "123.45 km" or "123.45 m")
  const infoPanel = page.getByTestId('info-panel');
  await expect(infoPanel).toBeVisible();

  // Poll for a measurement result. The format might vary, so we look for a number followed by a unit.
  // Common units: km, m, mi, ft.
  // We'll check the text content of the info panel for a pattern.
  await expect.poll(async () => {
    const text = await infoPanel.textContent();
    // Regex to find a number followed by a unit (km, m, mi, ft)
    // This is a heuristic; the actual implementation might vary.
    return /[\d.]+\s*(km|m|mi|ft)/i.test(text || '');
  }).toBeTruthy();
});
