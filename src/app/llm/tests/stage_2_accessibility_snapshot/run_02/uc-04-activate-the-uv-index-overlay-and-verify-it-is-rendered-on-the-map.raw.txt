import { test, expect } from '@playwright/test';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const layerSwitcher = page.getByTestId('layer-switcher');
  const mapContainer = page.getByTestId('map-container');
  const uvIndexCheckbox = layerSwitcher.getByRole('checkbox', { name: 'UV-Index', exact: true });

  await expect(layerSwitcher).toBeVisible();
  await expect(mapContainer).toBeVisible();
  await expect(uvIndexCheckbox).not.toBeChecked();

  const mapBefore = await mapContainer.screenshot();

  let captureRequests = false;
  const triggeredMapRequests: string[] = [];
  const successfulMapResponses: string[] = [];

  page.on('request', request => {
    if (!captureRequests) {
      return;
    }

    if (['image', 'xhr', 'fetch'].includes(request.resourceType())) {
      triggeredMapRequests.push(request.url());
    }
  });

  page.on('response', response => {
    if (!captureRequests) {
      return;
    }

    if (response.ok() && ['image', 'xhr', 'fetch'].includes(response.request().resourceType())) {
      successfulMapResponses.push(response.url());
    }
  });

  captureRequests = true;
  await uvIndexCheckbox.click({ force: true });

  await expect(uvIndexCheckbox).toBeChecked();
  await expect.poll(() => triggeredMapRequests.length > 0).toBe(true);
  await expect.poll(() => successfulMapResponses.length > 0).toBe(true);

  await expect.poll(async () => {
    const mapAfter = await mapContainer.screenshot();
    return mapAfter.equals(mapBefore);
  }).toBe(false);
});
