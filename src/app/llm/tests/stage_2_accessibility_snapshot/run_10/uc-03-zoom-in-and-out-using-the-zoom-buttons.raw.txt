import { test, expect } from '@playwright/test';

test('Use Case 3: Zoom in and out using the zoom buttons', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready and interactive
  const mapContainer = page.getByTestId('map-container');
  await expect(mapContainer).toBeVisible();

  // Helper to get current zoom level via the scale viewer text
  const getZoomLevel = async (p: typeof page) => {
    const scaleText = await p.getByTestId('scale-viewer').textContent();
    if (!scaleText) {
      return undefined;
    }
    // Format: "Current scale: 1 to 2739072"
    // Extract the denominator (scale denominator)
    const match = scaleText.match(/1 to (\d+)/);
    if (!match) {
      return undefined;
    }
    return parseInt(match[1], 10);
  };

  // Get initial zoom level (scale denominator)
  const initialZoom = await expect.poll(() => getZoomLevel(page)).toBeDefined();
  const initialScaleDenominator = initialZoom!;

  // Step 1: Click the 'Zoom in' button to increase the zoom level.
  // Zooming in means the scale denominator gets smaller (more detail).
  const zoomInButton = page.getByRole('button', { name: 'Zoom in map' });
  await expect(zoomInButton).toBeVisible();
  await zoomInButton.click();

  // Wait for the map to update and the scale to change
  const zoomedInScale = await expect.poll(() => getZoomLevel(page)).toBeDefined();
  expect(zoomedInScale).toBeLessThan(initialScaleDenominator);

  // Step 2: Click the 'Zoom out' button to decrease the zoom level.
  // Zooming out means the scale denominator gets larger (less detail).
  const zoomOutButton = page.getByRole('button', { name: 'Zoom out map' });
  await expect(zoomOutButton).toBeVisible();
  await zoomOutButton.click();

  // Wait for the map to update and the scale to change
  const zoomedOutScale = await expect.poll(() => getZoomLevel(page)).toBeDefined();
  expect(zoomedOutScale).toBeGreaterThan(zoomedInScale);
});
