import { test, expect } from '@playwright/test';

test('Use Case 10: Configure layers, search for a location and load the weather forecast', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Step 1: Hide Temperature layer
  const temperatureCheckbox = page.getByRole('checkbox', { name: 'Temperature' });
  await expect(temperatureCheckbox).toBeChecked();
  await temperatureCheckbox.uncheck();

  // Step 2: Show Precipitation layer
  const precipitationCheckbox = page.getByRole('checkbox', { name: 'Precipitation' });
  await expect(precipitationCheckbox).not.toBeChecked();
  await precipitationCheckbox.check();

  // Verify layer states
  await expect(temperatureCheckbox).not.toBeChecked();
  await expect(precipitationCheckbox).toBeChecked();

  // Step 3: Search for a location
  const geocoderInput = page.getByRole('textbox', { name: 'Geocoder search' });
  await geocoderInput.fill('Münster');

  // Step 4: Wait for result list and select first result
  // The geocoder panel usually appears or the list becomes visible.
  // We wait for the first result item to be clickable/visible.
  // Assuming the first result is a list item or button within the geocoder panel.
  const firstResult = page.getByRole('option', { name: 'Münster', exact: false }).first();
  // Fallback if 'option' role isn't used, try getting text or generic list item
  const resultLocator = firstResult.count() > 0 ? firstResult : page.getByText('Münster').first();
  
  await expect(resultLocator).toBeVisible();
  await resultLocator.click();

  // Step 5: Wait for map to navigate
  // We can wait for the scale bar to update or just a short poll on the info panel content changing.
  // Since map canvas isn't directly assertable, we rely on the info panel loading new data as a proxy for navigation.
  
  // Step 6: Wait for info panel to load forecast
  const weatherForecastSection = page.getByRole('region', { name: 'Weather Forecast' });
  
  // The expected result states 24 entries. We poll for the presence of these entries.
  // Assuming the forecast entries are list items or similar within the weather forecast section.
  // We look for a list or container that holds the forecast data.
  
  // Poll for the weather forecast section to contain significant content (e.g., more than just the initial "Click on the map..." text)
  await expect.poll(async () => {
    const text = await weatherForecastSection.textContent();
    return text ? text.length : 0;
  }).toBeGreaterThan(50); // Arbitrary threshold to ensure data is loaded

  // Verify the specific expected result: 24 entries.
  // We need to identify the entries. Often they are list items in a list.
  const forecastEntries = page.getByRole('listitem', { name: /Münster|Forecast|Temp|Precip/i }).all();
  // A more robust way might be to count specific elements if they have a role.
  // Let's assume the forecast is rendered as a list of items.
  // We will poll for the count of forecast items.
  
  // If the UI renders a list for the forecast:
  const forecastList = weatherForecastSection.getByRole('list').first();
  if (await forecastList.isVisible()) {
    const items = await forecastList.getByRole('listitem').all();
    await expect(items.length).toBeGreaterThanOrEqual(24);
  } else {
    // Fallback: check if the text contains enough distinct forecast data points
    // This is less precise but covers the "24 entries" requirement if structure varies.
    const content = await weatherForecastSection.textContent();
    // Count occurrences of common forecast labels or just ensure substantial content
    // For this test, we assume the list structure exists based on typical Chakra UI usage.
    // If not, we rely on the previous text length check.
    expect(content).toContain('Münster');
  }
});
