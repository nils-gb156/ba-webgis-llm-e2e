import { test, expect } from '@playwright/test';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // The layer switcher is visible by default according to preconditions.
  // Step 1: The user opens the base map selector in the layer switcher.
  // The combobox "Basemaps" is visible. Clicking it should open the dropdown.
  const basemapCombobox = page.getByRole('combobox', { name: 'Basemaps' });
  await expect(basemapCombobox).toBeVisible();
  await basemapCombobox.click();

  // Step 2: The user selects 'OpenStreetMap' as the base map.
  // The dropdown should contain an option or list item for OpenStreetMap.
  // We look for a list item or option with the text "OpenStreetMap".
  const osmOption = page.getByRole('option', { name: 'OpenStreetMap' });
  // If it's a listbox, options are role="option". If it's a custom dropdown, it might be a listitem.
  // Let's try role="option" first as it's standard for comboboxes.
  if (await osmOption.isVisible({ timeout: 2000 }).catch(() => false)) {
    await osmOption.click();
  } else {
    // Fallback to listitem if option is not found (custom implementation)
    const osmListItem = page.getByRole('listitem', { name: 'OpenStreetMap' });
    await expect(osmListItem).toBeVisible();
    await osmListItem.click();
  }

  // Expected results:
  // - The OpenStreetMap base map is selected.
  // - The Carto Light base map is no longer selected.
  await expect(basemapCombobox).toHaveText('OpenStreetMap');
});
