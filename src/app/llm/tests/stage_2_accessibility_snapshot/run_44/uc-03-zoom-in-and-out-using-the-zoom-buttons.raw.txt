import { test, expect } from '@playwright/test';

test('Use Case 3: Zoom in and out using the zoom buttons', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const zoomInButton = page.getByTestId('zoom-in-button');
  const zoomOutButton = page.getByTestId('zoom-out-button');
  const scaleViewer = page.getByTestId('scale-viewer');

  const parseScaleDenominator = (text: string | null): number | undefined => {
    if (!text) {
      return undefined;
    }

    const match = text.match(/1\s*(?::|to)\s*([\d.,\s]+)/i);
    if (!match) {
      return undefined;
    }

    const numericPart = match[1].replace(/[^\d]/g, '');
    const value = Number(numericPart);

    return Number.isNaN(value) ? undefined : value;
  };

  const getCurrentScaleDenominator = async (): Promise<number | undefined> => {
    const text = await scaleViewer.textContent();
    return parseScaleDenominator(text);
  };

  await expect(zoomInButton).toBeVisible();
  await expect(zoomOutButton).toBeVisible();
  await expect(scaleViewer).toBeVisible();

  const initialScaleDenominator = await expect
    .poll(async () => await getCurrentScaleDenominator())
    .toBeGreaterThan(0)
    .then(() => getCurrentScaleDenominator());

  await zoomInButton.click();

  await expect.poll(async () => await getCurrentScaleDenominator()).toBeLessThan(
    initialScaleDenominator as number
  );

  const zoomedInScaleDenominator = await getCurrentScaleDenominator();

  await zoomOutButton.click();

  await expect.poll(async () => await getCurrentScaleDenominator()).toBeGreaterThan(
    zoomedInScaleDenominator as number
  );
});
