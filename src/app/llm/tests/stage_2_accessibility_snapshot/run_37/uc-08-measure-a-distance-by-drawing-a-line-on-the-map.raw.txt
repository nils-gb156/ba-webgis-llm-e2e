import { test, expect } from '@playwright/test';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const map = page.getByTestId('map-container');
  const mapControlsPanel = page.getByTestId('map-controls-panel');
  const measurementToggle = page.getByTestId('measurement-toggle');
  const measurementDialog = page.getByRole('dialog', {
    name: /^Measure(?:ment)?$/i
  });
  const measurementHeading = page.getByRole('heading', {
    name: /^Measure(?:ment)?$/i
  });

  await expect(map).toBeVisible();
  await expect(mapControlsPanel).toBeVisible();
  await expect(measurementToggle).toBeVisible();

  const isMeasurementPanelVisible = async () =>
    (await measurementDialog.isVisible()) || (await measurementHeading.isVisible());

  const isMeasurementTogglePressed = async () =>
    (await measurementToggle.getAttribute('aria-pressed')) === 'true';

  if (!(await isMeasurementPanelVisible()) && !(await isMeasurementTogglePressed())) {
    await measurementToggle.click();
  }

  await expect
    .poll(async () => (await isMeasurementPanelVisible()) || (await isMeasurementTogglePressed()))
    .toBe(true);

  const mapBox = await map.boundingBox();
  expect(mapBox).not.toBeNull();
  if (!mapBox) {
    throw new Error('Map container has no bounding box.');
  }

  const firstPoint = {
    x: Math.round(mapBox.width * 0.35),
    y: Math.round(mapBox.height * 0.45)
  };
  const secondPoint = {
    x: Math.round(mapBox.width * 0.5),
    y: Math.round(mapBox.height * 0.55)
  };
  const thirdPoint = {
    x: Math.round(mapBox.width * 0.65),
    y: Math.round(mapBox.height * 0.4)
  };

  await map.click({ position: firstPoint });
  await map.click({ position: secondPoint });
  await map.dblclick({ position: thirdPoint });

  await expect
    .poll(async () => await mapControlsPanel.innerText())
    .toMatch(/(?:Length|Distance)[\s\S]*?\b\d+(?:[.,]\d+)?\s?(?:mm|cm|m|km)\b/i);
});
