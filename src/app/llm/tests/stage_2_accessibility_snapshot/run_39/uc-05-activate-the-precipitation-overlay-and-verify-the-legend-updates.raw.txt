import { test, expect } from '@playwright/test';

test('UC5: Activate the Precipitation overlay and verify the legend updates', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the app to load and the layer switcher to be visible
  await expect(page.getByTestId('layer-switcher')).toBeVisible();
  await expect(page.getByTestId('legend')).toBeVisible();

  // Step 1: Click the visibility toggle of the Precipitation overlay layer
  // The checkbox for "Precipitation" is inside the layer switcher list
  const precipitationCheckbox = page.getByRole('checkbox', { name: 'Precipitation' });
  await expect(precipitationCheckbox).toBeChecked({ checked: false });
  
  // Use force: true because Chakra UI renders the real input visually hidden
  await precipitationCheckbox.click({ force: true });

  // Verify the Precipitation overlay layer toggle is in the enabled (checked) state
  await expect(precipitationCheckbox).toBeChecked();

  // Step 2: View the legend and verify it displays an entry corresponding to the Precipitation layer
  // The legend is visible. We need to check if a new legend item for Precipitation appears.
  // Based on the accessibility tree, the legend contains headings for each active layer.
  // We expect to see a heading or text related to "Precipitation" in the legend.
  
  // Wait for the legend to update. Since we don't have a specific test id for the precipitation legend entry,
  // we look for the text "Precipitation" within the legend container.
  await expect(page.getByTestId('legend').getByText('Precipitation')).toBeVisible();
});
