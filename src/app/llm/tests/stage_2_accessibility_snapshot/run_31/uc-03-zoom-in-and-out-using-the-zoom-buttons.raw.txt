import { test, expect } from '@playwright/test';

test('Use Case 3: Zoom in and out using the zoom buttons', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready and initial layers to load
  await expect(page.getByTestId('map-container')).toBeVisible();
  await expect(page.getByRole('checkbox', { name: 'EUCOS Ground Stations' })).toBeChecked();

  // Capture initial zoom level
  const initialZoom = await page.evaluate(() => {
    // @ts-ignore - OpenLayers map is available on window
    return window.map?.getView()?.getZoom();
  });

  // Step 1: Click the 'Zoom in' button to increase the zoom level
  const zoomInButton = page.getByRole('button', { name: 'Zoom in map' });
  await zoomInButton.click();

  // Wait for zoom level to change
  await expect.poll(() => {
    return page.evaluate(() => {
      // @ts-ignore
      return window.map?.getView()?.getZoom();
    });
  }).toBeGreaterThan(initialZoom);

  // Capture zoom level after zooming in
  const zoomAfterIn = await page.evaluate(() => {
    // @ts-ignore
    return window.map?.getView()?.getZoom();
  });

  // Step 2: Click the 'Zoom out' button to decrease the zoom level
  const zoomOutButton = page.getByRole('button', { name: 'Zoom out map' });
  await zoomOutButton.click();

  // Wait for zoom level to change
  await expect.poll(() => {
    return page.evaluate(() => {
      // @ts-ignore
      return window.map?.getView()?.getZoom();
    });
  }).toBeLessThan(zoomAfterIn);

  // Verify that the zoom level after zooming out is lower than after zooming in
  const zoomAfterOut = await page.evaluate(() => {
    // @ts-ignore
    return window.map?.getView()?.getZoom();
  });

  expect(zoomAfterOut).toBeLessThan(zoomAfterIn);
});
