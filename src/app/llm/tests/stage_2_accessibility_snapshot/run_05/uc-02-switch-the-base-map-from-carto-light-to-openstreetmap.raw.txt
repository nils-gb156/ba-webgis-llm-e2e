import { test, expect } from '@playwright/test';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the application to load and the layer switcher to be visible/active
  await expect(page.getByRole('button', { name: 'Layer Switcher' })).toBeVisible();

  // Step 1: The user opens the base map selector in the layer switcher.
  // The layer switcher is already visible and active (pressed=true).
  // We need to find the combobox for "Basemaps".
  const basemapCombobox = page.getByRole('combobox', { name: 'Basemaps' });
  await expect(basemapCombobox).toBeVisible();

  // Step 2: The user selects 'OpenStreetMap' as the base map.
  // Open the combobox and select the option.
  await basemapCombobox.click();
  await page.getByRole('option', { name: 'OpenStreetMap' }).click();

  // Expected results:
  // - The OpenStreetMap base map is selected.
  // - The Carto Light base map is no longer selected.
  await expect(basemapCombobox).toHaveText('OpenStreetMap');
});
