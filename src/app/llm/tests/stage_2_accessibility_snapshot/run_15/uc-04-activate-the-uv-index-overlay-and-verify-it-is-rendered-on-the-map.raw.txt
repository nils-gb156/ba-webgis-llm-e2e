import { test, expect } from '@playwright/test';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');
  await page.waitForLoadState('networkidle');

  const layerSwitcher = page.getByTestId('layer-switcher');
  await expect(layerSwitcher).toBeVisible();

  const mapContainer = page.getByTestId('map-container');
  await expect(mapContainer).toBeVisible();

  const uvIndexCheckbox = page.getByRole('checkbox', { name: 'UV-Index', exact: true });
  await expect(uvIndexCheckbox).toBeVisible();
  await expect(uvIndexCheckbox).not.toBeChecked();

  const beforeMapScreenshot = await mapContainer.screenshot({ animations: 'disabled' });

  const isLikelyUvIndexTileRequest = (url: string): boolean => {
    const lowerUrl = url.toLowerCase();
    const mentionsUvIndex =
      lowerUrl.includes('uv-index') || lowerUrl.includes('uvindex') || lowerUrl.includes('uvi');
    const isLegendRequest =
      lowerUrl.includes('legend') || lowerUrl.includes('getlegendgraphic');
    const isStationRequest = lowerUrl.includes('station');
    const looksLikeMapImageRequest =
      lowerUrl.includes('getmap') ||
      lowerUrl.includes('bbox=') ||
      lowerUrl.includes('/tile') ||
      (lowerUrl.includes('width=') && lowerUrl.includes('height='));

    return mentionsUvIndex && !isLegendRequest && !isStationRequest && looksLikeMapImageRequest;
  };

  const capturedUvIndexRequests: string[] = [];
  page.on('request', (request) => {
    if (
      ['image', 'fetch', 'xhr'].includes(request.resourceType()) &&
      isLikelyUvIndexTileRequest(request.url())
    ) {
      capturedUvIndexRequests.push(request.url());
    }
  });

  const uvIndexTileResponsePromise = page.waitForResponse((response) => {
    const request = response.request();
    return (
      ['image', 'fetch', 'xhr'].includes(request.resourceType()) &&
      isLikelyUvIndexTileRequest(response.url()) &&
      response.ok()
    );
  });

  await uvIndexCheckbox.click({ force: true });
  await expect(uvIndexCheckbox).toBeChecked();

  await expect.poll(() => capturedUvIndexRequests[0] ?? '').toMatch(/uv-?index|uvindex|uvi/i);

  const uvIndexTileResponse = await uvIndexTileResponsePromise;
  expect(uvIndexTileResponse.ok()).toBeTruthy();

  await expect
    .poll(async () => {
      const currentMapScreenshot = await mapContainer.screenshot({ animations: 'disabled' });
      return currentMapScreenshot.equals(beforeMapScreenshot);
    })
    .toBe(false);
});
