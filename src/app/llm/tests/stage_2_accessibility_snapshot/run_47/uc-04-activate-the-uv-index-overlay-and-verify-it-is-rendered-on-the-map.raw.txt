import { test, expect } from '@playwright/test';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  const layerSwitcher = page.getByTestId('layer-switcher');
  const mapContainer = page.getByTestId('map-container');
  const uvIndexCheckbox = layerSwitcher.getByRole('checkbox', { name: 'UV-Index', exact: true });

  const isUvIndexTileUrl = (url: string) => {
    const lower = url.toLowerCase();
    const mentionsUvIndex =
      lower.includes('uv-index') ||
      lower.includes('uvindex') ||
      /(?:layers?|layer)=[^#]*\buvi\b/.test(lower) ||
      lower.includes('/uvi/');
    const looksLikeMapTile =
      lower.includes('getmap') ||
      lower.includes('service=wms') ||
      lower.includes('service=wmts') ||
      lower.includes('tilematrix=') ||
      lower.includes('tilerow=') ||
      lower.includes('tilecol=') ||
      lower.includes('bbox=');
    const isLegendGraphic = lower.includes('legend') || lower.includes('getlegendgraphic');

    return mentionsUvIndex && looksLikeMapTile && !isLegendGraphic;
  };

  await expect(layerSwitcher).toBeVisible();
  await expect(mapContainer).toBeVisible();
  await expect(uvIndexCheckbox).not.toBeChecked();

  await expect
    .poll(
      async () => {
        const first = await mapContainer.screenshot();
        const second = await mapContainer.screenshot();
        return first.equals(second);
      },
      { timeout: 15000 }
    )
    .toBe(true);

  const beforeScreenshot = await mapContainer.screenshot();
  expect(beforeScreenshot.byteLength).toBeGreaterThan(0);

  const uvIndexTileRequests: string[] = [];
  page.on('request', request => {
    if (request.resourceType() === 'image' && isUvIndexTileUrl(request.url())) {
      uvIndexTileRequests.push(request.url());
    }
  });

  const uvIndexTileResponsePromise = page.waitForResponse(
    response =>
      response.request().resourceType() === 'image' &&
      isUvIndexTileUrl(response.url()) &&
      response.ok(),
    { timeout: 15000 }
  );

  await uvIndexCheckbox.click({ force: true });
  await expect(uvIndexCheckbox).toBeChecked();

  const uvIndexTileResponse = await uvIndexTileResponsePromise;
  expect(uvIndexTileResponse.ok()).toBeTruthy();

  await expect.poll(() => uvIndexTileRequests.length, { timeout: 15000 }).toBeGreaterThan(0);

  await expect
    .poll(
      async () => {
        const afterScreenshot = await mapContainer.screenshot();
        return afterScreenshot.equals(beforeScreenshot);
      },
      { timeout: 15000 }
    )
    .toBe(false);
});
