import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle, isLayerRendered } from '../../../../map-model-helpers';

test('Use Case 9: Print the current map view as a PNG', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Precondition: At least one base map and one overlay layer are visible on the map.
  await expect.poll(() => getActiveBaseLayerTitle(page)).toBeTruthy();
  await expect.poll(() => isLayerRendered(page, 'EUCOS Ground Stations')).toBe(true);

  // Step 4: The user clicks the export/print button.
  // We capture the download before triggering the action.
  const [download] = await Promise.all([
    page.waitForEvent('download'),
    page.getByRole('button', { name: 'Print Map' }).click(),
  ]);

  // Step 1: The user clicks the 'Print Map' button in the toolbar to open the printing panel.
  // Step 2: The user enters a title for the printout.
  // Step 3: The user selects the PNG file format.
  // We wait for the print panel to appear and interact with it.
  const printPanel = page.getByRole('dialog', { name: /Print/ });
  await expect(printPanel).toBeVisible();

  await printPanel.getByLabel('Title').fill('My Map Print');
  await printPanel.getByRole('radio', { name: 'PNG' }).click();

  // Click the export/print button again to trigger the actual download.
  const [download2] = await Promise.all([
    page.waitForEvent('download'),
    printPanel.getByRole('button', { name: /Export|Print/ }).click(),
  ]);

  // Expected results: A PNG file containing the current map view is generated and downloaded.
  expect(download2.suggestedFilename()).toMatch(/\.png$/);
});
