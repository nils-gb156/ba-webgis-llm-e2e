import { test, expect } from '@playwright/test';

test('UC1: Show and hide the layer switcher via the toolbar button', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Step 1: Hide the layer switcher
    await page.getByRole('button', { name: 'Layer Switcher', exact: true }).click();

    // Expected result: layer switcher panel is no longer visible
    await expect(page.getByTestId('layer-switcher')).not.toBeVisible();

    // Step 2: Show the layer switcher again
    await page.getByRole('button', { name: 'Layer Switcher', exact: true }).click();

    // Expected result: layer switcher panel is visible again
    await expect(page.getByTestId('layer-switcher')).toBeVisible();
});
