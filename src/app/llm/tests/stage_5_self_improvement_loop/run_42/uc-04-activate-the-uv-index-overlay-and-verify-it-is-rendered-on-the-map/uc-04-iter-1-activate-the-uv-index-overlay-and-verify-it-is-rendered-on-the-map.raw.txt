import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready before interacting
  await expect(page.getByTestId('map-container')).toBeVisible();

  // Step 1: Click the visibility toggle of the UV-Index overlay layer
  // Use exact: true to distinguish "UV-Index" from "UV-Index Stations"
  const uvIndexCheckbox = page.getByRole('checkbox', { name: 'UV-Index', exact: true });
  await uvIndexCheckbox.click({ force: true });

  // Verify the UV-Index overlay layer toggle is in the enabled (checked) state
  await expect(uvIndexCheckbox).toBeChecked();

  // Step 2 & 3: Wait for and verify that the UV-Index overlay tiles are rendered on the map canvas
  await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
