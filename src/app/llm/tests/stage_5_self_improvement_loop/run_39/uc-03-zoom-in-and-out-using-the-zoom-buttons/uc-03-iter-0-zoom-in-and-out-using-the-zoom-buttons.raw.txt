import { test, expect } from '@playwright/test';
import { getMapZoomLevel } from '../../../../map-model-helpers';

test('UC3: Zoom in and out using the zoom buttons', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Capture the initial zoom level
  const initialZoom = await getMapZoomLevel(page);
  expect(initialZoom).toBeDefined();

  // Step 1: Click the 'Zoom in' button to increase the zoom level.
  await page.getByRole('button', { name: 'Zoom in map' }).click();

  // After clicking the 'Zoom in' button, the map zoom level is higher than before.
  await expect.poll(() => getMapZoomLevel(page)).toBeGreaterThan(initialZoom!);

  // Step 2: Click the 'Zoom out' button to decrease the zoom level.
  await page.getByRole('button', { name: 'Zoom out map' }).click();

  // After clicking the 'Zoom out' button, the map zoom level is lower than after zooming in.
  const zoomAfterOut = await getMapZoomLevel(page);
  expect(zoomAfterOut).toBeLessThan(initialZoom!);
});
