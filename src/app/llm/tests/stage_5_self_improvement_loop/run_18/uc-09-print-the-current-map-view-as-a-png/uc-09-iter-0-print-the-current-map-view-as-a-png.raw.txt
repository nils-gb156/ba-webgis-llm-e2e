import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle, isLayerRendered } from '../../../../map-model-helpers';

test('Use Case 9: Print the current map view as a PNG', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Precondition: Verify that at least one base map and one overlay layer are visible.
  await expect.poll(() => getActiveBaseLayerTitle(page)).toBeTruthy();
  await expect.poll(() => isLayerRendered(page, 'EUCOS Ground Stations')).toBe(true);

  // Step 1: Open the printing panel.
  await page.getByRole('button', { name: 'Print Map' }).click();

  // Expected result: The printing panel is visible.
  await expect(page.getByRole('dialog', { name: /Print Map/i })).toBeVisible();

  // Step 2: Enter a title for the printout.
  const dialog = page.getByRole('dialog', { name: /Print Map/i });
  await dialog.getByLabel('Title').fill('My Map Printout');

  // Step 3: Select the PNG file format.
  await dialog.getByRole('radio', { name: 'PNG' }).click();

  // Step 4: Trigger the export/print.
  const downloadPromise = page.waitForEvent('download');
  await dialog.getByRole('button', { name: 'Export' }).click();

  // Expected result: A PNG file is generated and downloaded.
  const download = await downloadPromise;
  await download.saveAs('/tmp/printout.png');
  expect(download.suggestedFilename()).toMatch(/\.png$/);
});
