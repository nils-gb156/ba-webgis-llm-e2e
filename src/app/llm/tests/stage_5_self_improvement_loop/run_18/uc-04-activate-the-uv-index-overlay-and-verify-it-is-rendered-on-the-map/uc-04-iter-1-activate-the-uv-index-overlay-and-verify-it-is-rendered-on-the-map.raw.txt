import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Step 1: Click the visibility toggle of the UV-Index overlay layer to show it.
  // The layer switcher is already visible per preconditions.
  // We force click because Chakra UI renders the real checkbox input visually hidden.
  // Use exact name to disambiguate from "UV-Index Stations".
  await page.getByRole('checkbox', { name: 'UV-Index', exact: true }).click({ force: true });

  // Verify the checkbox is now checked
  await expect(page.getByRole('checkbox', { name: 'UV-Index', exact: true })).toBeChecked();

  // Step 2: Wait for the map to load the layer tiles.
  // Step 2 Expected results: The UV-Index overlay tiles are rendered on the map canvas.
  await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
