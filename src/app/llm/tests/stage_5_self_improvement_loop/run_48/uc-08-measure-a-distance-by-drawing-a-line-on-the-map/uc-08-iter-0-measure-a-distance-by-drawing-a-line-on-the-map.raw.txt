import { test, expect } from '@playwright/test';
import { getMapCenter } from '../../../../map-model-helpers';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // 1. Activate the measurement tool
  const measurementToggle = page.getByTestId('measurement-toggle');
  await measurementToggle.click();

  // 2. Click several points on the map canvas to draw a line
  const mapContainer = page.getByTestId('map-container');
  const center = await expect.poll(() => getMapCenter(page)).toBeTruthy();
  // Click a few points around the center to draw a line
  await mapContainer.click({ position: { x: 300, y: 300 } });
  await page.waitForTimeout(300);
  await mapContainer.click({ position: { x: 400, y: 300 } });
  await page.waitForTimeout(300);
  await mapContainer.click({ position: { x: 400, y: 400 } });

  // 3. Double-click to finish the measurement
  await mapContainer.dblclick({ position: { x: 400, y: 400 } });

  // Expected results:
  // - The measurement panel is visible.
  // - The measurement panel displays a length value with a unit.
  const measurementPanel = page.getByTestId('map-controls-panel');
  await expect(measurementPanel).toBeVisible();

  // The measurement result typically appears in the info panel or a specific measurement result container.
  // Based on typical Open Pioneer trails behavior, the result appears in the info-panel.
  const infoPanel = page.getByTestId('info-panel');
  await expect(infoPanel).toBeVisible();

  // Check for a length value with a unit (e.g., "123 m" or "1.23 km")
  // The text might contain the length value and unit.
  await expect(infoPanel.locator('text=/\\d+\\.?\\d*\\s*(m|km|m²|ha)/')).toBeVisible();
});
