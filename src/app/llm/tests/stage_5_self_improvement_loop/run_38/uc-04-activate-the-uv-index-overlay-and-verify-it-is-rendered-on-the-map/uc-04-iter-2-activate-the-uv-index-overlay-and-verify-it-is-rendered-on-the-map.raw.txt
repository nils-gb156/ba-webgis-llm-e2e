import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Step 1: Click the visibility toggle of the UV-Index overlay layer to show it.
  // The layer switcher is already visible. The UV-Index checkbox is unchecked.
  // Use { exact: true } to disambiguate from "UV-Index Stations".
  // Chakra UI renders the real checkbox input visually hidden under a decorative
  // control element that intercepts pointer events, so we use { force: true }.
  const uvIndexCheckbox = page.getByRole('checkbox', { name: 'UV-Index', exact: true });
  await uvIndexCheckbox.click({ force: true });

  // Verify the toggle is now in the enabled (checked) state.
  await expect(uvIndexCheckbox).toBeChecked();

  // Step 2: Wait for the map to load the layer tiles.
  // Step 3: Verify the UV-Index overlay tiles are rendered on the map canvas.
  await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
