import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle } from '../../../../map-model-helpers';

test('Use Case 2: Switch the base map from Carto Light to OpenStreetMap', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // Wait for the map to be ready and the initial base layer to settle
    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('Carto Light');

    // Step 1: Open the base map selector in the layer switcher.
    // The combobox "Basemaps" is the selector.
    const basemapSelector = page.getByRole('combobox', { name: 'Basemaps' });
    await basemapSelector.click();

    // Step 2: Select 'OpenStreetMap' as the base map.
    // The dropdown should contain an option or list item with the text "OpenStreetMap".
    await page.getByRole('option', { name: 'OpenStreetMap' }).click();

    // Expected results: The OpenStreetMap base map is selected.
    await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('OpenStreetMap');
});
