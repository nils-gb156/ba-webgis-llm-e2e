import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../../map-model-helpers';

test('Use Case 5: Activate the Precipitation overlay and verify the legend updates', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Step 1: Activate the Precipitation overlay layer.
  // The layer switcher is already open. We find the checkbox by its accessible name.
  const precipitationCheckbox = page.getByRole('checkbox', { name: 'Precipitation' });
  await expect(precipitationCheckbox).toBeChecked({ checked: false });
  await precipitationCheckbox.click({ force: true });

  // Verify the layer toggle is now checked.
  await expect(precipitationCheckbox).toBeChecked();

  // Verify the layer is actually rendered on the map.
  await expect.poll(() => isLayerRendered(page, 'Precipitation')).toBe(true);

  // Step 2: View the legend and verify it contains an entry for the Precipitation layer.
  const legend = page.getByTestId('legend');
  await expect(legend).toContainText('Precipitation');
});
