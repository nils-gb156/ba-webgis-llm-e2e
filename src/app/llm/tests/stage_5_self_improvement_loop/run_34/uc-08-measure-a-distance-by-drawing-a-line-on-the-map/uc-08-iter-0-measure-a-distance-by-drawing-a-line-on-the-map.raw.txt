import { test, expect } from '@playwright/test';
import { getActiveBaseLayerTitle, getMapZoomLevel } from '../../../../map-model-helpers';

test('Use Case 8: Measure a distance by drawing a line on the map', async ({ page }) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  await expect(page.getByTestId('map-container')).toBeVisible();
  await expect(page.getByTestId('measurement-toggle')).toBeVisible();
  await expect.poll(() => getActiveBaseLayerTitle(page)).toBe('Carto Light');
  await expect.poll(() => getMapZoomLevel(page)).toBeGreaterThan(0);

  const controlsPanel = page.getByTestId('map-controls-panel');
  const measurementContainer = page
    .getByRole('dialog', { name: 'Measurement', exact: true })
    .or(page.getByRole('region', { name: 'Measurement', exact: true }))
    .or(controlsPanel);

  const measurementPanelIndicator = measurementContainer
    .getByRole('heading', { name: 'Measurement', exact: true })
    .or(measurementContainer.getByText(/^Length$/i))
    .or(measurementContainer.getByText(/click.*map/i));

  await page.getByTestId('measurement-toggle').click();
  await expect(measurementPanelIndicator.first()).toBeVisible();

  const mapContainer = page.getByTestId('map-container');
  await mapContainer.click({ position: { x: 520, y: 220 } });
  await mapContainer.click({ position: { x: 680, y: 300 } });
  await mapContainer.dblclick({ position: { x: 840, y: 380 } });

  await expect(
    measurementContainer.getByText(/\b\d[\d.,]*\s*(?:m|km)\b/).first()
  ).toBeVisible();
});
