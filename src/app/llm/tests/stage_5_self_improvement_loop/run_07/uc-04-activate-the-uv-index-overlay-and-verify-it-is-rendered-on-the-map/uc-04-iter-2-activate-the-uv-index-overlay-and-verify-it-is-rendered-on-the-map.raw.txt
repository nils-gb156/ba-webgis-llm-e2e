import { test, expect } from '@playwright/test';
import { getMapZoomLevel, isLayerRendered } from '../../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({
  page,
}) => {
  await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

  // Wait for the map to be ready before interacting
  await expect.poll(() => getMapZoomLevel(page)).toBeDefined();

  // Step 1: Click the visibility toggle of the UV-Index overlay layer
  // Scope to the "Operational layers" list to avoid ambiguity with "UV-Index Stations"
  await page
    .getByRole('list', { name: 'Operational layers' })
    .getByRole('checkbox', { name: 'UV-Index', exact: true })
    .click({ force: true });

  // Step 2: Wait for the map to load the layer tiles and verify state
  // Assert that the UV-Index overlay layer toggle is in the enabled (checked) state
  await expect(
    page.getByRole('list', { name: 'Operational layers' }).getByRole('checkbox', {
      name: 'UV-Index',
      exact: true,
    })
  ).toBeChecked();

  // Assert that the UV-Index overlay tiles are rendered on the map canvas
  await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
