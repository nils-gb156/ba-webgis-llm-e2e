import { test, expect } from '@playwright/test';
import { isLayerRendered } from '../../../../map-model-helpers';

test('Use Case 4: Activate the UV-Index overlay and verify it is rendered on the map', async ({ page }) => {
    await page.goto('http://localhost:5173/ba-webgis-llm-e2e/');

    // 1. The user clicks the visibility toggle of the UV-Index overlay layer to show it.
    // The name "UV-Index" is ambiguous because "UV-Index Stations" also exists.
    // Scope the lookup to the operational layers list to disambiguate.
    const operationalLayersList = page.getByRole('list', { name: 'Operational layers' });
    const uvIndexCheckbox = operationalLayersList.getByRole('checkbox', { name: 'UV-Index', exact: true });
    await uvIndexCheckbox.click({ force: true });

    // 2. The user waits for the map to load the layer tiles.
    // Expected results:
    // - The UV-Index overlay layer toggle is in the enabled (checked) state.
    await expect(uvIndexCheckbox).toBeChecked();

    // - The UV-Index overlay tiles are rendered on the map canvas.
    await expect.poll(() => isLayerRendered(page, 'UV-Index')).toBe(true);
});
